;= WRITE AT BASIC COMMAND ======================================

;READ AT drive,track,sector,addr(,sectors)
;WRITE AT drive,track,sector,addr(,sectors)
;VERIFY AT drive,track,sector,addr(,sectors)

;drive = drive number or 0 for last drive used
;addr  = address in memory from 16384 to 540671
;sectors = number of sectors

;Note: After each sector the memory pointer is increased by 512
;      for sectors with the sizes 128/256/1024 bytes, 'sectors'
;      must be 1 for correct operation.

write.bas:     CALL xat.sub
WMSAD:         LD   HL,wr.buff
               JR   xsad

;= READ AT BASIC COMMAND =======================================

read.bas:      CALL xat.sub
RMSAD:         LD   HL,rd.buff
               JR   xsad

;= VERIFY AT BASIC COMMAND =====================================

veri.bas:      CALL xat.sub
VMSAD:         LD   HL,ve.buff
               JR   xsad

;- WRITE SECTOR AT DE hook code --------------------------------

HWSAD:         LD   HL,wr.buff
               JR   rwsad

;- READ SECTOR AT DE hook code ---------------------------------

;A= drive number
;DE= track sector
;HL= memory address 16384 to 65535

HRSAD:         LD   HL,rd.buff

rwsad:         PUSH HL
               LD   HL,(hk.hl)
               LD   A,H
               AND  192
               JR   Z,rep30        ;error if HL<16384
               SUB  64
               RLCA
               RLCA
               LD   (hk.bc),A      ;select page at CD
               LD   A,H
               AND  63
               LD   (hk.hl+1),A
               LD   HL,1
               LD   (hk.ix),HL
               POP  HL

;- rd/wr/ve/ hk.IX sectors drive A at DE to page C offset HL -

xsad:          PUSH HL             ;=rd buff wr buff or ve buff
               LD   A,(hk.a)
bckup.rd:      CALL set.drive.a
               CALL res.buf
               IN   A,(251)
               LD   (L16686),A
               LD   BC,(hk.ix)     ;sectors
               LD   A,B
               OR   C              ;error if sectors=0
               JR   Z,rep30
               LD   DE,(hk.de)     ;track sector
               LD   HL,(hk.hl)     ;set memory address
               SET  7,H
               LD   A,(hk.bc)
               OUT  (251),A
sad1:
               LD   (L30735),HL    ;set buf ad at memory
               POP  HL
               PUSH HL
               PUSH BC
               CALL &0005          ;jp (hl)
               INC  H              ;HL=HL+512
               INC  H
               CALL inc.sct
               JR   NZ,sad2        ;jr no trk change
               INC  D
               BIT  6,H
               JR   Z,sad2         ;jrr no page change

               RES  6,H
               IN   A,(251)
               INC  A
               OUT  (251),A

sad2:          POP  BC
               DEC  BC
               LD   A,B
               OR   C
               JR   NZ,sad1
               POP  BC             ;drop service call

               LD   A,(L16686)
               OUT  (251),A
               RET

;= '.' POINT command ===========================================

;.filename,parameters
;loads file 'filename' from command record into current active
;screen. By adding a seperator parameters can be added.
;Parameters are skipped during syntax. During runtime CHADD
;points to 1st chr after the seperator to be used by command.
;code.

point.bas:     LD   B,11
               LD   DE,uifa+1
pntb0:         CALL tst.end
               JR   Z,pntb1        ;end of file name

               LD   (DE),A
               INC  DE
               CALL nxt.chr
               DJNZ pntb0
               JP   rep29          ;error name >10chrs

pntb1:         CALL exp.end        ;exit if syntax checking

pntb2:         INC  B              ;pad name with spaces
               INC  B
               INC  B
               CALL sp.to.de

               LD   HL,(com.record)
               LD   (record.t),HL  ;select command record
               SCF
               CALL exp.rcd

               CALL res.buf
               CALL L22875         ;find file
               JP   NZ,rep107      ;file not found
               LD   A,(HL)         ;HL=dir entry
               AND  31
               CP   19             ;code file
               JP   NZ,rep94       ;wrong file
               LD   (HL),A

;error if file start not => 32768

               PUSH HL
               LD   BC,236
               ADD  HL,BC
               CALL pt.scrof
               LD   C,6
               ADD  HL,BC
               DEC  (HL)
               CALL pt.scrof
               INC  (HL)
               POP  HL
               CALL point.sub      ;move file info to hdr/hdl

;Blank screen for command

               LD   A,&DB          ;OUT (),A >> IN A,()
               CALL blanking.on
               LD   A,128
               OUT  (254),A

;MAKE sure the file is loaded & run correctly

               LD   A,149          ;'LOAD' token
               CALL poke           ;as current command
               DEFW &5B74
               LD   HL,&7FFF       ;Make sure that ROM 1 is
               SET  6,(HL)         ;switched in by RST 8 code
               JP   L23310         ;hdr2hdr hdl2hdl,exit&load

               IN   A,(252)
               AND  31
               ADD  C              ;C=start page
               LD   (HL),A         ;set scr as start
               RET

;next record entry

               LD   A,L              ;HL=next record entry
               ADD  B              ;B=16 or 0
               LD   L,A
               JR   NC,$+3
               INC  H
               DEC  DE
               LD   A,D
               OR   E
               JR   Z,rcd.nf       ;error not found
               DEC  C
               JR   NZ,frec2       ;next entry same sector

;next record list sector

               POP  HL
               INC  HL             ;next record list sector
               JR   frec1

rcd.nf:        POP  HL             ;restore base ?
               RET

; - select temp record routine -

sel.record:    LD   DE,(record.t)

sel.rcd:       LD   HL,(last.record)
               SCF
               DEC  DE              ;record 0 = 65535
               SBC  HL,DE           ;test valid record number
               JP   C,rep81
               LD   A,&D0           ;=normal record size
               JR   NZ,nwrc0        ;jr not last record
               LD   A,(last.recs)
nwrc0:         LD   (L16531),A      ;set drive tracks
               EX   DE,HL           ;HL=record no.
               LD   BC,1600
               CALL mult16
               LD   A,C             ;AHL=sectors*record

;adjust base sectors for boot/record list

               LD   BC,(base)
               ADD  HL,BC
               ADC  0

sel.base:      LD   C,0
               LD   DE,(hd.sct)     ;DE= hd*cylinders
               CALL div24
               LD   (st.c),DE       ;base cylinder
               LD   DE,(max.s)
               LD   D,C             ;D=0
               DEC  E               ;CDE =sectors per track
               CALL div24
               LD   A,E            ;
               LD   (st.h),A       ;store base head
               LD   A,L
               INC  A
               LD   (st.s),A       ;store base sector
               RET

;-16 bit multiply routine (result is 24bit)

mult16:        EX   DE,HL            ;BCHL=HL*BC
               LD   A,B
               LD   B,16
               LD   HL,0

clcm1:         ADD  HL,HL
               RL   C
               RLA
               JR   NC,clcm2
               ADD  HL,DE
               JR   NC,$+3
               INC  C
clcm2:         DJNZ clcm1
               LD   B,A
               RET

;- 24 bit division using 16bit input CDE=HL/DE,AHL=HL MOD DE -

div16:         XOR  A
               LD   C,A

;-24 bit division  routine CDE=AHL/CDE,AHL= AHL MOD CDE -

div24:         PUSH HL
               LD   HL,0
               EXX
               POP  HL
               LD   C,A          ;CHL'=AHL
               XOR  A
               LD   B,24

dv1:           ADC  HL,HL
               RL   C
               EXX
               ADC  HL,HL
               ADC  A
               SBC  HL,DE
               SBC  C
               JR   NC,$+4
               ADD  HL,DE
               ADC  C
               CCF
               EXX
               DJNZ dv1
               ADC  HL,HL
               RL   C
               EX   DE,HL          ;CDE=result
               EXX
               PUSH HL
               EXX
               POP  HL             ;AHL =mod
               RET

;- convert drive data to max sectors -

drive.size:    LD   D,A            ;A=drive data
               LD   E,1

;- conver trk D sct E to logical sector -

conv.de:       LD   A,D
               ADD  A
               JR   NC,$+4         ;jr not side 2
               ADD  160            ;adjust for side 2
               LD   L,A
               RLA
               AND  1
               LD   H,A            ;HL=2* log track*2
               LD   A,E
               DEC  A
               ADD  L
               LD   C,A            ;BC=HL+E-1
               LD   A,H
               ADC  0
               LD   B,A
               ADD  HL,HL
               ADD  HL,HL
               ADD  HL,BC          ;HL=tracks * 10 + sector-1
               RET

;= RESTORE DEVICE COMMAND ======================================

rest.bas:      LD   C,240          ;DEVICE
               CALL exp.c
               CALL exp.end
               LD   HL,(last.record)
               LD   A,H
               OR   L
               JP   Z,rep103       ;no hard disk
               LD   A,2
               CALL rcd.da         ;set device d2
               CALL def.rec        ;restore D2 data by selecting
               CALL sel.record     ;default record
               JP   sres           ;reset drive logic

;= DEVICE BASIC COMMAND ========================================

;DEVICE OFF(,n) puts HD in stand by mode (,n: sets APD)
;DEVICE ON(,n)  gives the HD a kick in the butt
;               puts HD in idle mode (,n:sets APD)
;DEVICE STOP    will sing the hard disk asleep

dev.bas:       LD   C,137            ;'OFF'
               CALL skip.c
               JR   NZ,dev.on
               CALL tst.end
               JR   Z,def.off      ;jr DEVICE OFF
               LD   B,&E2          ;stand by apd
dev.apd:       PUSH BC
               CALL exp1num
               PUSH HL
               CALL exp.end
               POP  HL
               CALL hd.busy
               LD   A,&F2
               OUT  (&F5),A
               LD   A,L
               OUT  (&F7),A        ;apd number
               POP  AF
               JR   hd.chk

def.off:       CALL exp.end

hd.stand.by:   LD   A,&E0          ;stand by immediate
               JR   hd.chk

dev.on:        LD   C,222          ;'ON'
               CALL skip.c
               JR   NZ,dev.stop
               CALL tst.end
               LD   B,&E3          ;idle apd
               JR   NZ,dev.apd
               CALL exp.end


hd.idle:       LD   A,&E1          ;idle immediate
               JR   hd.chk

dev.stop:      LD   C,177          ;'STOP'
               CALL exp.c
               CALL exp.end
               CALL hd.sleep
               XOR  A
               LD   (L16531),A     ;no d2 data for no such drive
               INC  A
               JP   rcd.da         ;set device d1

hd.sleep:      LD   A,&E6          ;HD sleep command

hd.chk:        CALL chk.hd.ex

;- send command to hard disk -

wr.com:        EX   AF,AF'
               CALL hd.busy
               LD   A,&F7
               OUT  (&F5),A        ;select register
               EX   AF,AF'
               OUT  (&F7),A
               CALL hd.busy
io.out:        LD   A,&FF          ;chip select off
               OUT  (&F5),A
               RET

;- reset hard disk -

sres:          LD   A,&EE
               OUT  (&F5),A
               OUT  (&F7),A         ;Software reset
               LD   A,&FA
               OUT  (&F7),A
               JR   io.out

hd.ready:      CALL hd.busy
               OR   &16            ;drq and index
               CP   &B6
               RET  Z
               JR   hd.ready

;- wait while busy and test ESC -

hd.busy:       LD   A,&F7          ;get busy stat in carry
               OUT  (&F5),A
hdbs:          CALL tst.esc
               IN   A,(&F6)
               IN   A,(&F7)
               ADD  A
               JR   C,hdbs
               RET

wait.drq:      LD   A,&F7          ;wait for data request
               OUT  (&F5),A
wdrq:          IN   A,(&F6)
               IN   A,(&F7)
               AND  %00001001
               JR   Z,wdrq
               RRA
               JR   C,hd.error     ;command error
               LD   A,&F0          ;select data reg
               OUT  (&F5),A        ;note Carry is error
               LD   BC,&FEF7
               RET

hd.error:      CALL sres
               JP   rep85

rep83:         CALL L20133
               DEFB 83             ;sector not found

;= Seek track D sector E from offset'base' =====================

hd.seek:       LD   A,D            ;check valid track and sector
               ADD  A              ;first to prevent corruption
               CP   2*79+1         ;in next record
               JR   NC,rep83       ;tracks >79 invalid
               LD   A,E
               AND  A
               JR   Z,rep83          ;sector 0 invalid
               CP   11               ;sector >10 invalid
               JR   NC,rep83
               PUSH DE
               PUSH HL
               CALL conv.de
               LD   BC,(hd.sct)    ;A=HL/(hd*sct)
               XOR  A
               INC  A
               SBC  HL,BC
               JR   NC,$-3
               ADD  HL,BC
               DEC  A
               LD   D,A            ;D=cylinder offset

               LD   BC,(max.s)    ;A=HL/(sct)
               DEC  C
               XOR  A
               LD   B,A
               INC  A
               SBC  HL,BC
               JR   NC,$-3
               ADD  HL,BC
               DEC  A
               LD   E,A            ;E=heads offset

;add page ofset

               LD   A,L            ;sector offset
               LD   HL,(st.s)
               ADD  L
               CP   H
               JR   C,nsb
               DEC  H
               SUB  H
               INC  E
nsb:           EX   AF,AF'         ;save sector

               LD   A,E            ;heads
               LD   HL,(st.h)
               ADD  L
               CP   H
               JR   C,nhb
               SUB  H              ;heads start at 0
               INC  D
nhb:           CALL hd.mix         ;mix hd drive to E
               LD   B,0
               LD   C,D            ;cyls
               LD   HL,(st.c)
               ADD  HL,BC          ;HL=cyl E=head
                                   ;A'=sector
;-send C H S to HD -

               LD   BC,&01F7
               LD   A,&F2
               OUT  (&F5),A
               OUT  (C),B
               INC  A
               OUT  (&F5),A

               OUT  (C),A
               EX   AF,AF'
               INC  A
               OUT  (&F5),A
               OUT  (C),L
               INC  A
               OUT  (&F5),A
               OUT  (C),H
               INC  A
               OUT  (&F5),A
               OUT  (C),E
               LD   A,&FF
               OUT  (&F5),A
               POP  HL
               POP  DE
               JP   adbor

               XOR  D                ; 1 * side

               CALL format.BC

               CALL format.BC
               LD   BC,&03F5       ;3  * F5

               CALL format.BC

               CALL format.C

               CALL format.BC
               DEC  L
               JR   NZ,fmtrk2      ;next sector

               CALL format.BC      ;768* 4E
               CALL format.BC
               CALL format.BC
               RET

               CALL prt.a

;- print space -

prt.sp:        LD   A," "

;- print character A -

prt.a:         PUSH AF
               CALL sys.call
               DEFW &0010          ;prt chr all regs saved
               POP  AF
               RET

prt.quote:     LD   A,""
               JR   prt.a

L22222:        CALL single.dr
               RET  NZ
               CALL prt.cls
               DEFB 13
               DEFM "Enter source disk press any ke"
               DEFB "y"+128
               JR   get.key

L22194:        CALL single.dr
               RET  NZ
               CALL prt.cls
               DEFM "Enter target disk press any ke"
               DEFB "y"+128

;- wait key and cls lower screen -

get.key:       CALL sys.call
               DEFW &0169
               JR   NC,get.key
               PUSH AF             ;save key
L22209:        CALL sys.call
               DEFW &0169
               JR   C,L22209
               CALL cls.low
               POP  AF             ;A=KEY
               AND  223
               CP   "Y"
               RET

cls.low:       PUSH IX
               CALL sys.call       ;cls lower screen
               DEFW &0151
               POP  IX
               RET


cls.stream:    LD   A,(stream)     ;select stream and cls
               CP   2              ;if #2
               JR   NZ,sel.stream

cls:           XOR  A
               CALL sys.call
               DEFW &014E          ;clear entire screen
stream2:       LD   A,2
sel.stream:    CALL sys.call       ;select S
               DEFW &0112
               RET

;-request (Yes No Continue End) sub for erase copy -

erase.req:     CALL prt.cls
               DEFM "Erase "
               DEFB ""+128
               JR   request

copy.req:      CALL prt.cls
               DEFM "Copy "
               DEFB ""+128

request:       CALL prt.name       ;print file name"(Y/N/C/E)
               CALL prt.mess
               DEFB ""
               DEFM " (Y/N/C/E) "
               DEFB "?"+128

reqst0:        CALL get.key
               RET  Z              ;ret 'Y' pressed
               CP   "C"
               JR   Z,reqst1       ;'C' cancel request
               CP   "E"
               RET  NZ             ;ret for 'N' or other key
               POP  BC             ;'E' end command by unstack
               JP   search.end     ;error if no file handled

reqst1:        LD   HL,L30732
               RES  7,(HL)
               RET

yes.no:        CALL prt.mess
               DEFB ""
               DEFM " (Y/N"
               DEFB ")"+128
               CALL beep
               JP   get.key
                                   ;Z if 'Y' pressed
;- BEEP -

beep:          PUSH HL
               PUSH DE
               PUSH BC
               PUSH IX
               LD   HL,874
               LD   DE,133
               CALL sys.call
               DEFW &016F
               POP  IX
               POP  BC
               POP  DE
               POP  HL
               RET

single.dr:     LD   A,(dev.no)
               LD   B,A
               LD   A,(L16722)
               CP   B
               RET  NZ
               DEC  A
               RET                   ;Z floppy disk only

;= error handler ===============================================

L20133:        CALL blanking.off
               CALL rest.bor
               LD   HL,(nmi.sp)
               LD   A,H
               OR   L
               JR   Z,L20145       ;jr return to SAM
               LD   SP,HL
               RET                   ;to NMI

;- poke track sector in error message -

L20145:        LD   HL,er.trk      ;track digits to error mess
               PUSH HL
               LD   A,D
               CALL dig.thl
               LD   HL,er.sct
               LD   A,E            ;add  sector digits
               CALL dig.thl
               POP  HL
               PUSH HL
               LD   DE,er.ts2      ;copy to sct. error message
               LD   BC,er.tsl
               LDIR
               POP  HL
               LD   DE,er.ts3      ;copy to sct. lost message
               LD   C,er.tsl
               LDIR

               CALL dpeek
               DEFW &5A97
               CALL dpoke
               DEFW &5AA3
               XOR  A
               LD   (L30732),A     ;clear flags
               LD   E,A            ;no action
               POP  HL
               LD   A,(HL)         ;get error code after call
               LD   SP,(ent.sp)
               RET

;- convert number A to digits at HL -

dig.thl:       EX   AF,AF'
               SCF                   ;signal leading zero = space
               LD   BC,100
               CALL dig.a
               LD   BC,10
               CALL dig.a
               EX   AF,AF'
               LD   B,A
               JR   diga2

dig.a:         EX   AF,AF'
diga1:         INC  B
               SUB  C
               JR   NC,diga1
               ADD  C
               EX   AF,AF'
               DEC
